\section{Organelos celulares: Estructura, características y funciones}

\subsection{Núcleo}
\begin{itemize}
    \item Estructura: Envoltura nuclear, poros nucleares, nucleoplasma, nucléolo
    \item Características: Contiene ADN, controla la expresión génica
    \item Función: Almacenamiento y transmisión de información genética
\end{itemize}

\subsection{Mitocondrias}
\begin{itemize}
    \item Estructura: Membrana externa, membrana interna (crestas), matriz
    \item Características: ADN propio, se replican por fisión
    \item Función: Producción de ATP por respiración aeróbica
\end{itemize}

\subsection{Retículo endoplásmico}
\begin{itemize}
    \item Estructura: Red de membranas interconectadas
    \item Tipos: RE rugoso (con ribosomas) y RE liso (sin ribosomas)
    \item Función: Síntesis de proteínas (rugoso) y lípidos (liso)
\end{itemize}

\subsection{Aparato de Golgi}
\begin{itemize}
    \item Estructura: Cisternas aplanadas apiladas
    \item Características: Polaridad (cis y trans)
    \item Función: Modificación, empaquetamiento y distribución de proteínas
\end{itemize}

\subsection{Lisosomas}
\begin{itemize}
    \item Estructura: Vesículas con enzimas hidrolíticas
    \item Características: pH ácido interno (4.5-5.0)
    \item Función: Digestión intracelular y reciclaje
\end{itemize}

\subsection{Peroxisomas}
\begin{itemize}
    \item Estructura: Vesículas con enzimas oxidativas
    \item Características: Contienen catalasa
    \item Función: Oxidación de ácidos grasos y detoxificación
\end{itemize}

\subsection{Vacuolas}
\begin{itemize}
    \item Estructura: Vesículas grandes (en vegetales)
    \item Características: Membranosas, contienen agua y solutos
    \item Función: Almacenamiento y mantenimiento de turgencia
\end{itemize}

\subsection{Ribosomas}
\begin{itemize}
    \item Estructura: Subunidades (60S y 40S en eucariotas)
    \item Características: ARNr + proteínas
    \item Función: Síntesis de proteínas
\end{itemize}

\section{Citosol}
\begin{itemize}
    \item Componente líquido del citoplasma
    \item Contiene agua, sales, enzimas, metabolitos
    \item Función: Medio para reacciones metabólicas
\end{itemize}

\section{Inclusiones celulares}
\begin{itemize}
    \item Gotas lipídicas: Almacenamiento de lípidos
    \item Glucógeno: Reserva energética
    \item Pigmentos: Melanina, hemosiderina
\end{itemize}
